# Themes with External Upload Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Users upload JSON theme files in Settings → Appearance; themes are stored on the server, shared by all devices, and applied per device as CSS variable overrides.

**Architecture:** `magpie/themes.py` validates and stores `<data dir>/themes/<id>.json`; three routes in `main.py` list/upload/delete; the frontend caches the chosen theme in `localStorage` and applies it through `data-theme="custom"` + `data-base` + an injected `<style id="magpie-theme">`.

**Tech Stack:** Python 3 / FastAPI / pytest (TestClient); vanilla JS in `magpie/static/`.

**Spec:** `docs/superpowers/specs/2026-10-13-themes-design.md`

## Global Constraints

- Theme `id`: `[a-z0-9-]{1,40}`. `name`: 1–40 printable chars. `base`: `light` or `dark`.
- `colors` keys, and only these: `bg surface surface-2 text muted border accent accent-soft on-accent ok err warn warn-soft`. Unknown keys rejected; missing keys fall back to the base palette.
- Color values: `#rgb`, `#rrggbb`, `#rrggbbaa`, or `rgb()/rgba()/hsl()/hsla()/oklch()` containing only digits, `.`, `%`, `,`, `/`, spaces, `-`, `deg`. No `url()`, `var()`, `;`, `}`.
- File size at most 8192 bytes (`themes.MAX_BYTES`).
- Write routes call `require_settings_access`; they are NOT gated by `MAGPIE_ALLOW_PLUGIN_INSTALL`.
- Choice stored per device: `localStorage["magpie.theme"] = "custom:<id>"`, theme data cached in `localStorage["magpie.theme.data"]`.
- `sw.js` already skips `/api/` requests (`sw.js:52`), so `/api/themes` is not cached; no change needed there.

## Review Focus

- Custom theme with `base: light` on an OS in dark mode must stay light (the `prefers-color-scheme: dark` block must not apply). Pinned in Task 3.
- Color value like `red;}body{display:none` or `url(http://x)` must be rejected, not stored. Pinned in Task 1.
- Upload that is not JSON, not UTF-8, or a JSON array/`null` gives 422, never 500. Pinned in Task 1/2.
- Theme chosen on this device then deleted from another device: next load falls back to System, no broken page. Pinned in Task 3.
- Re-uploading the same `id` replaces the theme and the active device picks up the new colors. Pinned in Task 2/3.

```python
# magpie/themes.py
"""Uploaded colour themes: a small JSON file of CSS-variable colours, validated hard (it ends up in a <style>),
stored as <data dir>/themes/<id>.json and shared by every device."""

import json
import os
import re
from pathlib import Path

MAX_BYTES = 8 * 1024
COLOR_KEYS = ("bg", "surface", "surface-2", "text", "muted", "border", "accent", "accent-soft",
              "on-accent", "ok", "err", "warn", "warn-soft")
ID_RE = re.compile(r"^[a-z0-9-]{1,40}$")
_HEX = re.compile(r"^#(?:[0-9a-fA-F]{3}|[0-9a-fA-F]{6}|[0-9a-fA-F]{8})$")
_FUNC = re.compile(r"^(?:rgb|rgba|hsl|hsla|oklch)\((?:[0-9.%,/ \-]|deg){1,60}\)$")


class ThemeError(Exception):
    def __init__(self, message: str, status: int = 422):
        super().__init__(message)
        self.status = status


def _folder(data_dir: Path) -> Path:
    return Path(data_dir) / "themes"


def validate(raw: bytes) -> dict:
    if len(raw) > MAX_BYTES:
        raise ThemeError(f"The theme file is too large (the limit is {MAX_BYTES // 1024} KB).")
    try:
        data = json.loads(raw.decode("utf-8"))
    except (UnicodeDecodeError, ValueError):
        raise ThemeError("The theme file isn't valid JSON.")
    if not isinstance(data, dict):
        raise ThemeError("The theme file must be a JSON object.")
    theme_id, name, base, colors = data.get("id"), data.get("name"), data.get("base"), data.get("colors")
    if not isinstance(theme_id, str) or not ID_RE.match(theme_id):
        raise ThemeError("\"id\" must be 1-40 characters: lowercase letters, digits and dashes.")
    if not isinstance(name, str) or not 1 <= len(name.strip()) <= 40 or not name.isprintable():
        raise ThemeError("\"name\" must be 1-40 characters of plain text.")
    if base not in ("light", "dark"):
        raise ThemeError("\"base\" must be \"light\" or \"dark\".")
    if not isinstance(colors, dict):
        raise ThemeError("\"colors\" must be an object.")
    for key, value in colors.items():
        if key not in COLOR_KEYS:
            raise ThemeError(f"Unknown color \"{key}\". Allowed: {', '.join(COLOR_KEYS)}.")
        if not isinstance(value, str) or not (_HEX.match(value) or _FUNC.match(value)):
            raise ThemeError(f"\"{key}\" must be a color like #1a2b3c or rgb(26 43 60).")
    return {"id": theme_id, "name": name.strip(), "base": base, "colors": dict(colors)}


def list_themes(data_dir: Path) -> list[dict]:
    out = []
    for path in sorted(_folder(data_dir).glob("*.json")):
        try:
            theme = validate(path.read_bytes())
        except (OSError, ThemeError):
            continue
        if theme["id"] == path.stem:
            out.append(theme)
    return out


def save(data_dir: Path, theme: dict) -> None:
    folder = _folder(data_dir)
    folder.mkdir(parents=True, exist_ok=True)
    path = folder / f"{theme['id']}.json"
    tmp = path.with_suffix(".json.tmp")
    tmp.write_text(json.dumps(theme, ensure_ascii=False, indent=2), encoding="utf-8")
    os.replace(tmp, path)


def remove(data_dir: Path, theme_id: str) -> None:
    if not isinstance(theme_id, str) or not ID_RE.match(theme_id):
        raise ThemeError("No such theme.", 404)
    try:
        (_folder(data_dir) / f"{theme_id}.json").unlink()
    except FileNotFoundError:
        raise ThemeError("No such theme.", 404)
```

```python
        require_settings_access(request)
        try:
            themes.remove(settings.data_dir, theme_id)
        except themes.ThemeError as e:
            raise HTTPException(e.status, str(e))
        except OSError as e:
```

- [ ] **Step 5: app.js appearance code.** Replace `applyTheme`, `installIcons`, `setTheme` and extend `appearanceHtml` (keep `getTheme`, `THEME_COLORS`, `prefersDark`):

```js
const THEME_COLOR_RE = /^(#[0-9a-f]{3}|#[0-9a-f]{6}|#[0-9a-f]{8}|(rgb|rgba|hsl|hsla|oklch)\([0-9a-z.%,\/ -]*\))$/i;
let customThemes = [];   // uploaded themes, from the server

// The cached data of the chosen "custom:<id>" theme, else null (also when the cache belongs to another theme).
function customThemeData(theme = getTheme()) {
  if (!theme.startsWith("custom:")) return null;
  let d = null;
  try { d = JSON.parse(localStorage.getItem("magpie.theme.data") || "null"); } catch {}
  return d && `custom:${d.id}` === theme && (d.base === "light" || d.base === "dark") ? d : null;
}

// "light" or "dark" when the theme forces one, null when it follows the device.
function themeBase(theme) {
  return theme === "light" || theme === "dark" ? theme : customThemeData(theme)?.base || null;
}

function applyTheme(theme) {
  const root = document.documentElement;
  const custom = customThemeData(theme);
  const base = themeBase(theme);
  if (custom) { root.dataset.theme = "custom"; root.dataset.base = custom.base; }
  else {
    delete root.dataset.base;
    if (base) root.dataset.theme = base; else delete root.dataset.theme;
  }
  let style = document.getElementById("magpie-theme");
  if (custom) {
    if (!style) { style = document.createElement("style"); style.id = "magpie-theme"; document.head.append(style); }
    const vars = Object.entries(custom.colors || {})
      .filter(([k, v]) => /^[a-z0-9-]+$/.test(k) && typeof v === "string" && THEME_COLOR_RE.test(v))
      .map(([k, v]) => `--${k}:${v}`).join(";");
    style.textContent = `:root[data-theme="custom"][data-base]{${vars}}`;
  } else style?.remove();
  // Browser/PWA chrome colour: forced themes override both media-specific values.
  document.querySelectorAll('meta[name="theme-color"]').forEach((m) => {
    const scheme = (m.media.match(/(light|dark)/) || [])[1] || "light";
    const own = custom?.colors?.bg;
    m.content = own && THEME_COLOR_RE.test(own) ? own : THEME_COLORS[base || scheme];
  });
  installIcons(theme);
}

function installIcons(theme = getTheme()) {
  const base = themeBase(theme);
  const dark = base === "dark" || (base === null && !!prefersDark?.matches);
  const set = (selector, href) => { const el = document.querySelector(selector); if (el && el.getAttribute("href") !== href) el.setAttribute("href", href); };
  set('link[rel="manifest"]', dark ? "/static/manifest-dark.webmanifest" : "/static/manifest.webmanifest");
  set('link[rel="apple-touch-icon"]', dark ? "/static/icons/apple-touch-icon-dark.png" : "/static/icons/apple-touch-icon.png");
}
prefersDark?.addEventListener?.("change", () => installIcons());

function setTheme(theme) {
  try {
    if (theme.startsWith("custom:")) {
      const t = customThemes.find((c) => `custom:${c.id}` === theme);
      if (!t) return;
      localStorage.setItem("magpie.theme", theme);
      localStorage.setItem("magpie.theme.data", JSON.stringify({ id: t.id, base: t.base, colors: t.colors }));
    } else {
      localStorage.removeItem("magpie.theme.data");
      if (theme === "system") localStorage.removeItem("magpie.theme");
      else localStorage.setItem("magpie.theme", theme);
    }
  } catch {}
  applyTheme(theme);
  document.querySelectorAll("[data-theme-choice]").forEach((b) =>
    b.setAttribute("aria-pressed", String(b.dataset.themeChoice === theme)));
}

// Uploaded themes are kept on the server and shared by every device; which one is used is chosen per device.
async function loadThemes() {
  try { customThemes = await api("/api/themes"); } catch { return; }   // offline: keep the cached theme
  const current = getTheme();
  if (current.startsWith("custom:")) {
    if (customThemes.some((t) => `custom:${t.id}` === current)) setTheme(current);   // refresh the cached colours
    else setTheme("system");                                                          // deleted elsewhere
  }
  refreshAppearance();
}

function refreshAppearance() {
  const section = document.querySelector("[data-theme-list]")?.closest(".settings-section");
  if (section) section.innerHTML = `<h3>Appearance</h3>${appearanceHtml()}`;
}

async function uploadTheme(file) {
  const body = new FormData();
  body.append("file", file);
  try {
    const t = await apiWithSetup("/api/themes", { method: "POST", body });
    toast(`Added the theme “${t.name}”.`);
  } catch (e) { return toast(`Couldn't add the theme: ${e.message}`); }
  await loadThemes();
}

async function removeTheme(id) {
  try { await apiWithSetup(`/api/themes/${encodeURIComponent(id)}`, { method: "DELETE" }); }
  catch (e) { return toast(`Couldn't remove the theme: ${e.message}`); }
  await loadThemes();
}
```
